cmake_minimum_required(VERSION 3.25)
project(InputAlignExternal VERSION 0.2.0 LANGUAGES CXX)

option(IAX_BUILD_MOD "Build the Windows Geode bridge" ${WIN32})
option(IAX_BUILD_TESTS "Build protocol and input-gate tests" ON)

# The documented MIT simulator fork is independent of Geode and can be tested on Linux.
add_library(gdsim SHARED worker/branch.cpp)
target_include_directories(gdsim PRIVATE include)
target_compile_features(gdsim PRIVATE cxx_std_17)
set_target_properties(gdsim PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(WIN32)
    set_property(TARGET gdsim PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(MSVC)
    target_compile_options(gdsim PRIVATE /fp:strict)
else()
    target_compile_options(gdsim PRIVATE -ffp-contract=off -fno-fast-math)
endif()

if(IAX_BUILD_TESTS)
    enable_testing()
    add_executable(iax_candidates_tests tests/candidates.cpp)
    target_include_directories(iax_candidates_tests PRIVATE include)
    target_compile_features(iax_candidates_tests PRIVATE cxx_std_17)
    add_test(NAME candidates COMMAND iax_candidates_tests)
    add_executable(iax_gate_tests tests/gate.cpp)
    target_include_directories(iax_gate_tests PRIVATE include)
    target_compile_features(iax_gate_tests PRIVATE cxx_std_17)
    add_test(NAME gate COMMAND iax_gate_tests)
    add_executable(iax_mechanics_tests tests/native_mechanics.cpp)
    target_include_directories(iax_mechanics_tests PRIVATE include)
    target_compile_features(iax_mechanics_tests PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(iax_mechanics_tests PRIVATE /fp:strict)
        set_property(TARGET iax_mechanics_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    else()
        target_compile_options(iax_mechanics_tests PRIVATE -ffp-contract=off -fno-fast-math)
    endif()
    add_test(NAME mechanics COMMAND iax_mechanics_tests)
    add_executable(iax_deadline_tests tests/deadline.cpp)
    target_include_directories(iax_deadline_tests PRIVATE include)
    target_compile_features(iax_deadline_tests PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(iax_deadline_tests PRIVATE /fp:strict)
    else()
        target_compile_options(iax_deadline_tests PRIVATE -ffp-contract=off -fno-fast-math)
    endif()
    add_test(NAME deadline COMMAND iax_deadline_tests)
endif()

if(IAX_BUILD_MOD)
    if(NOT WIN32 OR NOT DEFINED ENV{GEODE_SDK})
        message(FATAL_ERROR "The bridge requires Windows x64 and GEODE_SDK=Geode 5.11.2 SDK")
    endif()
    add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
    add_library(inputalign_external SHARED src/main.cpp src/Runtime.cpp src/Panel.cpp)
    target_include_directories(inputalign_external PRIVATE include src)
    target_compile_features(inputalign_external PRIVATE cxx_std_23)
    target_compile_definitions(inputalign_external PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(inputalign_external PRIVATE ws2_32)
    setup_geode_mod(inputalign_external DONT_INSTALL LINK_TYPE PRIVATE)
endif()
